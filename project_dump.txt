# sql-widget

Виджет SQL-заданий для платформы совместного обучения
(Miro-подобной). Преподаватель создаёт карточку задания
(формулировка, баллы, подсказки с пенальти), назначает студента;
студент пишет SQL-запрос, отправляет на проверку и может попросить
помощь по выделенным строкам; преподаватель при проверке отмечает
ошибочные строки и выставляет баллы.

Модуль спроектирован максимально лёгким: виджет не исполняет SQL
сам и не хранит состояние — всё это делает платформа через
интерфейсы-порты.

## Установка

```sh
npm install sql-widget
```

Peer-зависимости: `react` >= 18.

## Использование

```tsx
import { SqlWidget } from "sql-widget";
import type { SqlExecutor, TaskCard } from "sql-widget";

const executor: SqlExecutor = {
  // платформа сама решает, где выполнять запросы
  execute: async (sql) => {
    const res = await fetch("/api/sql", {
      method: "POST",
      body: JSON.stringify({ sql }),
    });
    return res.json(); // { columns, rows }
  },
};

<SqlWidget
  card={card}
  role="student"
  solution={solution}
  executor={executor}
  onSubmit={(sql) => api.submit(card.id, sql)}
  onUseHint={(hintId) => api.useHint(card.id, hintId)}
  onSelectionRequest={(request) => api.help(request)}
  onReview={(result) => api.review(card.id, result)}
  onCardChange={(next) => api.saveCard(next)}
/>;
```

Виджет управляемый: состояние карточки хранит платформа.
Итоговый балл считается как `maxScore` минус пенальти раскрытых
подсказок (`calculateScore`, тоже экспортируется).

## Mock-режим (без платформы)

Точка входа `sql-widget/mock` содержит mock-реализации портов:

```ts
import {
  createSqlJsExecutor, // SQL на sql.js (SQLite в WASM)
  MOCK_CARDS, // готовые карточки заданий
  MOCK_SCHEMA, // таблицы SQLite для них
  MockPlatform, // in-memory хранилище состояния
} from "sql-widget/mock";
```

`createSqlJsExecutor` принимает `wasmUrl` — URL файла
`sql-wasm.wasm` (в сборках Vite:
`import url from "sql.js/dist/sql-wasm.wasm?url"`).

## Демо-стенд

```sh
npm install
npm run demo
```

Откроется страница с двумя панелями (преподаватель и студент)
и общим состоянием: отправка решения, подсказки со списанием
баллов, проверка и запросы по выделенным строкам видны сразу
с обеих ролей.

export function App() {
  const platform = useMemo(() => new MockPlatform(MOCK_CARDS), []);
  useSyncExternalStore(platform.subscribe, platform.getVersion);
  const [cardId, setCardId] = useState(MOCK_CARDS[1].id);
  const [executor, setExecutor] = useState<SqlExecutor | null>(null);
  const [studentSel, setStudentSel] = useState<SelectionRange | null>(null);
  const [teacherSel, setTeacherSel] = useState<SelectionRange | null>(null);

  useEffect(() => {
    let cancelled = false;
    createSqlJsExecutor({ wasmUrl: sqlWasmUrl, schema: MOCK_SCHEMA })
      .then((created) => {
        if (!cancelled) setExecutor(created);
      })
      .catch(() => undefined);
    return () => {
      cancelled = true;
    };
  }, []);

  const card = platform.getCard(cardId);
  if (!card || !executor) {
    return (
      <Box p="4">
        <Text>Загрузка sql.js…</Text>
      </Box>
    );
  }
  const solution = platform.getSolution(cardId);
  const isAfterSubmit =
    card.status === "submitted" || card.status === "reviewed";

  function createCard() {
    const id = `card-${Date.now().toString(36)}`;
    platform.updateCard({
      id,
      prompt: "",
      maxScore: 10,
      hints: [],
      assigneeId: null,
      status: "draft",
      usedHintIds: [],
    });
    setCardId(id);
  }

  return (
    <Box p={{ initial: "3", sm: "5" }} maxWidth="1400px" mx="auto">
      <Flex direction="column" gap="3">
        <Flex align="center" gap="3" wrap="wrap">
          <Heading size="5">SQL Widget — демо полного UX-flow</Heading>
          <Badge color={STATUS_COLORS[card.status]} size="2">
            {STATUS_LABELS[card.status]}
          </Badge>
        </Flex>
        <Text size="2" color="gray">
          {STEP_HINTS[card.status]}
        </Text>
        <Flex align="center" gap="3" wrap="wrap">
          <Select.Root value={cardId} onValueChange={setCardId}>
            <Select.Trigger className="card-select" />
            <Select.Content>
              {platform.getCards().map((item) => (
                <Select.Item key={item.id} value={item.id}>
                  {item.prompt.slice(0, 40) || `Черновик ${item.id}`}
                </Select.Item>
              ))}
            </Select.Content>
          </Select.Root>
          <Button onClick={createCard}>Новое задание</Button>
        </Flex>
        <Flex
          direction={{ initial: "column", lg: "row" }}
          gap="4"
          align="start"
        >
          <Box className="screen-col">
            <Screen title="Экран преподавателя — Платформа">
              <SqlWidget
                card={card}
                solution={solution}
                executor={executor}
                role="teacher"
                highlight={isAfterSubmit ? null : studentSel}
                onCardChange={(next) => platform.updateCard(next)}
                onSubmit={() => undefined}
                onUseHint={() => undefined}
                onReturnForRevision={(comment) =>
                  platform.returnForRevision(cardId, comment)
                }
                onReview={(result) => platform.review(cardId, result)}
                onSelectionChange={setTeacherSel}
              />
            </Screen>
          </Box>
          <Box className="screen-col">
            <Screen title="Экран студента — Платформа">

                solution={solution}
                executor={executor}
                role="student"

                onSubmit={(sqlText) =>

                }

                onReview={() => undefined}
                onSelectionChange={setStudentSel}
              />
            </Screen>
          </Box>
        </Flex>
      </Flex>
    </Box>
  );
}

=== FILE: docs/SPEC.txt ===
Это модуль для платформы совместного обучения.
Платформа совместного обучения представляет собой что-то похожее на Miro.
Смысл модуля в предоставлении простого API для расширения платформы, 
который позволяет добавить на платформу виджет для исполнения SQL-кода.

Зачем нужен виджет: интерактивное обучение на онлайн-платформе, смысл 
который в возможности онлайн-взаимодействия преподавателя и студента 
по предмету SQL базы данных.
Смысл в следующем: преподаватель создает карточку с заданием, где
указывает формулировку задачи, сколько баллов за решение, 
а также может добавить список подсказок, каждый из которых понижает итоговые балл; 
после чего преподаватель назначает студента, который должен решить задачу.
После того, как преподаватель заполнил карточку, студент приступает к решению задачи, 
нажимая на карточку. Здесь он может писать SQL-запрос, после чего отправлять его для
проверки решения. Если у него возникают какие-то трудности, то он может выделить строки
запроса и попросить преподавателя помочь ему, нажав соответствующую кнопку. 
Аналогичным образом, при проверке работы, преподаватель может выделять неправильные строки.

Детали реализации.
Модуль проектировался как можно более легкий во всех смыслах: количестве
и сложности реализации зависимостей, минималитичном API (чем меньше -- тем 
сложнее сломать и проще интегрировать).
Модуль сделан на React и TypeScript (так как он используется и в целевой платформе), 
весь модуль упаковывается в npm-пакет для стандартной последующей интеграции.
В качестве стилей используется крошечная библиотека компонентов, позволяющая организовать
единый интерфейс на всех платформах.

import type { CardStatus } from "./types.js";

const TRANSITIONS: Record<CardStatus, CardStatus[]> = {
  draft: ["assigned"],
  assigned: ["in_progress"],
  in_progress: ["submitted"],
  submitted: ["reviewed", "in_progress"],
  reviewed: [],
};

/**
 * Допустим ли переход карточки из статуса `from` в статус `to`.
 * Цикл: draft → assigned → in_progress → submitted → reviewed;
 * из submitted возможен возврат на доработку (in_progress).
 */
export function canTransition(from: CardStatus, to: CardStatus): boolean {
  return TRANSITIONS[from].includes(to);
}

import { describe, expect, test } from "vitest";
import { canTransition } from "./lifecycle.js";
import { calculateScore } from "./score.js";
import { getSelectedText, toSelectionRange } from "./selection.js";
import type { TaskCard } from "./types.js";

function makeCard(overrides: Partial<TaskCard> = {}): TaskCard {
  return {
    id: "card-1",
    prompt: "Выбери все строки",
    maxScore: 10,
    hints: [
      { id: "h1", text: "Первая", penalty: 2 },
      { id: "h2", text: "Вторая", penalty: 3 },
    ],
    assigneeId: "student-1",
    status: "assigned",
    usedHintIds: [],
    ...overrides,
  };
}

describe("calculateScore", () => {
  test("без подсказок — полный балл", () => {
    expect(calculateScore(makeCard())).toBe(10);
  });

  test("вычитает пенальти раскрытых подсказок", () => {
    const card = makeCard({ usedHintIds: ["h1", "h2"] });
    expect(calculateScore(card)).toBe(5);
  });

  test("не опускается ниже нуля", () => {
    const card = makeCard({
      maxScore: 3,
      usedHintIds: ["h1", "h2"],
    });
    expect(calculateScore(card)).toBe(0);
  });

  test("игнорирует неизвестные id подсказок", () => {
    const card = makeCard({ usedHintIds: ["nope"] });
    expect(calculateScore(card)).toBe(10);
  });
});

describe("canTransition", () => {
  test("линейный жизненный цикл", () => {
    expect(canTransition("draft", "assigned")).toBe(true);
    expect(canTransition("assigned", "in_progress")).toBe(true);
    expect(canTransition("in_progress", "submitted")).toBe(true);
    expect(canTransition("submitted", "reviewed")).toBe(true);
    expect(canTransition("submitted", "in_progress")).toBe(true);
  });

  test("запрещает пропуски и возвраты", () => {
    expect(canTransition("draft", "submitted")).toBe(false);
    expect(canTransition("reviewed", "in_progress")).toBe(false);
    expect(canTransition("assigned", "assigned")).toBe(false);
  });
});

describe("getSelectedText / toSelectionRange", () => {
  const sql = "SELECT a\nFROM t\nWHERE a > 1";

  test("одна строка", () => {
    const text = getSelectedText(sql, { fromLine: 2, toLine: 2 });
    expect(text).toBe("FROM t");
  });

  test("диапазон строк", () => {
    const text = getSelectedText(sql, { fromLine: 1, toLine: 2 });
    expect(text).toBe("SELECT a\nFROM t");
  });

  test("toSelectionRange нормализует порядок и 0-based ввод", () => {
    expect(toSelectionRange(2, 0)).toEqual({
      fromLine: 1,
      toLine: 3,
    });
  });
});

export interface SqlJsExecutorOptions {
  /** URL файла sql-wasm.wasm; без него sql.js ищет его сам. */
  wasmUrl?: string;
  /** Скрипт создания таблиц, выполняется при запуске. */
  schema?: string;
}

/**
 * Mock-исполнитель SQL на sql.js (реальный SQLite в WASM).
 * Каждая база живёт в памяти вкладки браузера.
 */
export async function createSqlJsExecutor(
  options: SqlJsExecutorOptions = {},
): Promise<SqlExecutor> {
  const SQL = await initSqlJs(
    options.wasmUrl
      ? { locateFile: () => options.wasmUrl as string }
      : undefined,
  );
  const db: Database = new SQL.Database();
  if (options.schema) db.run(options.schema);
  return {
    async execute(sqlText: string): Promise<SqlQueryResult> {
      const results = db.exec(sqlText);
      const first = results[0];
      if (!first) return { columns: [], rows: [] };
      return {
        columns: first.columns,
        rows: first.values as (string | number | null)[][],
      };
    },
  };
}

  CardStatus,

  TaskCard,
} from "../types.js";

const ORDER: CardStatus[] = [
  "draft",
  "assigned",
  "in_progress",
  "submitted",
  "reviewed",
];

/** Задержка, имитирующая сетевой запрос, мс. */
const LATENCY_MS = 300;

type Listener = () => void;

/**
 * In-memory имитация платформы: хранит карточки и решения,
 * обновляется асинхронно.
 */
export class MockPlatform {
  private cards: TaskCard[];
  private solutions = new Map<string, Solution>();
  private listeners = new Set<Listener>();
  private version = 0;

  constructor(cards: TaskCard[]) {
    this.cards = cards.map((card) => structuredClone(card));
  }

  subscribe = (listener: Listener): (() => void) => {
    this.listeners.add(listener);
    return () => this.listeners.delete(listener);
  };

  getVersion = (): number => this.version;

  getCards(): TaskCard[] {
    return this.cards.map((card) => structuredClone(card));
  }

  getCard(cardId: string): TaskCard | undefined {
    const card = this.cards.find((c) => c.id === cardId);
    return card ? structuredClone(card) : undefined;
  }

  getSolution(cardId: string): Solution | null {
    const solution = this.solutions.get(cardId);
    return solution ? structuredClone(solution) : null;
  }

  /** Мгновенно: задержка сделала бы ввод текста мучительным. */
  updateCard(card: TaskCard): void {
    const index = this.cards.findIndex((c) => c.id === card.id);
    if (index < 0) {
      this.cards.push(structuredClone(card));
    } else {
      this.cards[index] = structuredClone(card);
    }
    this.version += 1;
    this.listeners.forEach((listener) => listener());
  }

  useHint(cardId: string, hintId: string): void {
    this.apply(() => {
      const card = this.find(cardId);
      if (!card || card.usedHintIds.includes(hintId)) return;
      card.usedHintIds.push(hintId);
      this.advance(card, "in_progress");
    });
  }

  startWork(cardId: string): void {
    this.apply(() => {
      const card = this.find(cardId);
      if (card) this.advance(card, "in_progress");
    });
  }

  submit(
    cardId: string,
    sql: string,
    highlight: SelectionRange | null = null,
  ): void {
    this.apply(() => {
      const card = this.find(cardId);
      if (!card) return;
      this.advance(card, "in_progress");
      this.advance(card, "submitted");
      this.solutions.set(cardId, {
        cardId,
        sql,
        review: null,
        highlight,
        revisionComment: null,
      });
    });
  }

  review(cardId: string, result: ReviewResult): void {
    this.apply(() => {
      const card = this.find(cardId);
      const solution = this.solutions.get(cardId);
      if (!card || !solution) return;
      this.advance(card, "reviewed");
      this.solutions.set(cardId, {
        cardId,
        sql: solution.sql,
        review: result,
      });
    });
  }

  /** Возврат решения на доработку: submitted → in_progress. */
  returnForRevision(cardId: string, comment: string): void {
    this.apply(() => {
      const card = this.find(cardId);
      const solution = this.solutions.get(cardId);
      if (!card || !solution) return;
      if (!canTransition(card.status, "in_progress")) return;
      card.status = "in_progress";
      this.solutions.set(cardId, {
        ...solution,
        revisionComment: comment || null,
      });
    });
  }

  private find(cardId: string): TaskCard | undefined {
    return this.cards.find((card) => card.id === cardId);
  }

  private advance(card: TaskCard, to: CardStatus): void {
    let index = ORDER.indexOf(card.status);
    const target = ORDER.indexOf(to);
    while (index < target) {
      const next = ORDER[index + 1];
      if (!canTransition(card.status, next)) return;
      card.status = next;
      index += 1;
    }
  }

  private apply(mutate: () => void): void {
    setTimeout(() => {
      mutate();
      this.version += 1;
      this.listeners.forEach((listener) => listener());
    }, LATENCY_MS);
  }
}

/**
 * Итоговый балл за карточку: базовый балл минус пенальти
 * за раскрытые подсказки, но не ниже нуля.
 */
export function calculateScore(card: TaskCard): number {
  const penalty = card.hints
    .filter((hint) => card.usedHintIds.includes(hint.id))
    .reduce((sum, hint) => sum + hint.penalty, 0);
  return Math.max(0, card.maxScore - penalty);
}

 */
export function toSelectionRange(
  fromLine: number,
  toLine: number,
): SelectionRange {
  return {

    toLine: Math.max(fromLine, toLine) + 1,
  };
}

import { useEffect, useRef } from "react";
import { toSelectionRange } from "../selection.js";
import type { SelectionRange } from "../types.js";
import { applyHighlight, highlightExtension } from "./highlight.js";

interface SqlEditorProps {
  value: string;
  readOnly?: boolean;
  /** Строки, подсвеченные второй ролью. */
  highlight?: SelectionRange | null;
  onChange?(value: string): void;
  onSelectRange?(range: SelectionRange | null): void;
}

/**
 * Редактор SQL на CodeMirror 6: нумерация строк, выделение
 * строк мышью и подсветка строк второй роли.
 */
export function SqlEditor(props: SqlEditorProps) {
  const { value, readOnly = false, highlight = null } = props;
  const host = useRef<HTMLDivElement>(null);
  const view = useRef<EditorView | null>(null);
  const callbacks = useRef(props);
  callbacks.current = props;

  useEffect(() => {
    if (!host.current) return;
    const editor = new EditorView({
      doc: callbacks.current.value,
      extensions: [
        basicSetup,
        sql(),
        highlightExtension,
        EditorView.editable.of(!readOnly),
        EditorView.updateListener.of((update) => {
          const { onChange, onSelectRange } = callbacks.current;
          if (update.docChanged) {
            onChange?.(update.state.doc.toString());
          }
          if (update.selectionSet && onSelectRange) {
            const sel = update.state.selection.main;
            const doc = update.state.doc;
            onSelectRange(
              toSelectionRange(
                doc.lineAt(sel.from).number - 1,
                doc.lineAt(sel.to).number - 1,
              ),
            );
          }
        }),
      ],
      parent: host.current,
    });
    view.current = editor;
    return () => {
      editor.destroy();
      view.current = null;
    };
  }, [readOnly]);

  useEffect(() => {
    const editor = view.current;
    if (!editor) return;
    const current = editor.state.doc.toString();
    if (current !== value) {
      editor.dispatch({
        changes: { from: 0, to: current.length, insert: value },
      });
    }
    applyHighlight(editor, highlight);
  }, [value, highlight]);

  return <div className="sqlw-editor" ref={host} />;
}

import { useEffect, useState } from "react";
import { calculateScore } from "../score.js";
import type { SqlQueryResult, SqlWidgetProps } from "../types.js";
import { CardEditor } from "./CardEditor.js";
import { ResultsTable } from "./ResultsTable.js";
import { SqlEditor } from "./SqlEditor.js";
import "./styles.css";

const STATUS_LABELS: Record<string, string> = {
  draft: "Черновик",
  assigned: "Назначено",
  in_progress: "В работе",
  submitted: "На проверке",
  reviewed: "Проверено",
};

/**
 * Виджет SQL-задания: карточка, редактор запроса с общей
 * подсветкой строк, подсказки, проверка и возврат на доработку.
 */
export function SqlWidget(props: SqlWidgetProps) {
  const { card, role, solution } = props;
  const [draftSql, setDraftSql] = useState(solution?.sql ?? "");
  const [result, setResult] = useState<SqlQueryResult | null>(null);
  const [error, setError] = useState<string | null>(null);
  const [running, setRunning] = useState(false);
  const [reviewScore, setReviewScore] = useState(calculateScore(card));
  const [reviewComment, setReviewComment] = useState("");

  useEffect(() => {
    setDraftSql(solution?.sql ?? "");
    setResult(null);
    setError(null);
  }, [card.id]);

  const usedHints = card.usedHintIds.join(",");
  useEffect(() => {
    setReviewScore(calculateScore(card));
  }, [card.maxScore, usedHints, card.status]);

  const isTeacher = role === "teacher";
  const canSolve =
    !isTeacher && (card.status === "assigned" || card.status === "in_progress");
  const canReview = isTeacher && card.status === "submitted";

  async function runQuery(sqlText: string) {
    setRunning(true);
    setError(null);
    try {
      setResult(await props.executor.execute(sqlText));
    } catch (err) {
      setResult(null);
      setError(err instanceof Error ? err.message : String(err));
    } finally {
      setRunning(false);
    }
  }

  if (isTeacher && card.status === "draft") {
    return (
      <div className="sqlw">
        <CardEditor
          card={card}
          onChange={(next) => props.onCardChange?.(next)}
        />
      </div>
    );
  }

  const editorSql = canSolve ? draftSql : (solution?.sql ?? "");

  return (
    <div className="sqlw">
      <header className="sqlw-header">
        <span className="sqlw-status">{STATUS_LABELS[card.status]}</span>
        <span className="sqlw-score">
          Баллы: {calculateScore(card)} из {card.maxScore}
        </span>
      </header>
      <p className="sqlw-prompt">{card.prompt}</p>

      {!isTeacher && card.hints.length > 0 && (
        <div className="sqlw-hints">
          {card.hints.map((hint) =>
            card.usedHintIds.includes(hint.id) ? (
              <p key={hint.id} className="sqlw-hint-used">
                {hint.text}
              </p>
            ) : (
              <button
                key={hint.id}
                type="button"
                disabled={!canSolve}
                onClick={() => props.onUseHint(hint.id)}
              >
                Подсказка (−{hint.penalty} б.)
              </button>
            ),
          )}
        </div>
      )}

      <SqlEditor
        value={editorSql}
        readOnly={!canSolve}
        highlight={props.highlight ?? solution?.highlight ?? null}
        onChange={setDraftSql}
        onSelectRange={(range) => props.onSelectionChange?.(range)}
      />

      <div className="sqlw-actions">
        <button
          type="button"
          disabled={running || !editorSql.trim()}
          onClick={() => void runQuery(editorSql)}
        >
          {running ? "Выполняется…" : "Выполнить"}
        </button>
        {canSolve && (
          <button
            type="button"
            disabled={!draftSql.trim()}
            onClick={() => props.onSubmit(draftSql)}
          >
            Отправить на проверку
          </button>
        )}
      </div>

      <ResultsTable result={result} error={error} />

      {canReview && (
        <div className="sqlw-review">
          <label>
            <span>Оценка</span>
            <input
              type="number"
              min={0}
              max={card.maxScore}
              value={reviewScore}
              onChange={(e) => setReviewScore(Number(e.target.value) || 0)}
            />
          </label>
          <label>
            <span>Комментарий</span>
            <textarea
              rows={2}
              placeholder="Комментарий"
              value={reviewComment}
              onChange={(e) => setReviewComment(e.target.value)}
            />
          </label>
          <div className="sqlw-actions">
            <button
              type="button"
              onClick={() =>
                props.onReview({
                  score: reviewScore,
                  comment: reviewComment,
                })
              }
            >
              Оценить
            </button>
            {props.onReturnForRevision && (
              <button
                type="button"
                onClick={() => props.onReturnForRevision?.(reviewComment)}
              >
                Вернуть на доработку
              </button>
            )}
          </div>
        </div>
      )}

      {canSolve && solution?.revisionComment && (
        <div className="sqlw-revision">
          <strong>Комментарий преподавателя:</strong>
          <p>{solution.revisionComment}</p>
        </div>
      )}

      {solution?.review && (
        <div className="sqlw-review-result">
          <strong>Итог: {solution.review.score} б.</strong>
          {solution.review.comment && <p>{solution.review.comment}</p>}
        </div>
      )}
    </div>
  );
}

import { RangeSetBuilder, StateEffect, StateField } from "@codemirror/state";
import { Decoration, EditorView } from "@codemirror/view";
import type { DecorationSet } from "@codemirror/view";
import type { SelectionRange } from "../types.js";

const mark = Decoration.mark({ class: "sqlw-highlight" });

const setHighlight = StateEffect.define<{
  from: number;
  to: number;
} | null>();

const highlightField = StateField.define<DecorationSet>({
  create: () => Decoration.none,
  update(value, tr) {
    value = value.map(tr.changes);
    for (const effect of tr.effects) {
      if (!effect.is(setHighlight)) continue;
      if (!effect.value) {
        value = Decoration.none;
        continue;
      }
      const builder = new RangeSetBuilder<Decoration>();
      builder.add(effect.value.from, effect.value.to, mark);
      value = builder.finish();
    }
    return value;
  },
  provide: (field) => EditorView.decorations.from(field),
});

/**
 * Расширение CodeMirror для подсветки строк, выделенных
 * второй ролью в общем рабочем пространстве.
 */
export const highlightExtension = [highlightField];

/**
 * Подсветить диапазон строк (номера с 1) в открытом редакторе.
 */
export function applyHighlight(
  view: EditorView,
  range: SelectionRange | null,
): void {
  if (!range) {
    view.dispatch({ effects: setHighlight.of(null) });
    return;
  }
  const doc = view.state.doc;
  const fromLine = Math.min(Math.max(range.fromLine, 1), doc.lines);
  const toLine = Math.min(Math.max(range.toLine, 1), doc.lines);
  view.dispatch({
    effects: setHighlight.of({
      from: doc.line(fromLine).from,
      to: doc.line(toLine).to,
    }),
  });
}
